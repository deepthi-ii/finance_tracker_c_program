cmake_minimum_required(VERSION 3.16)
project(RupeeSense C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
find_package(PkgConfig REQUIRED)
pkg_check_modules(MHD REQUIRED libmicrohttpd)
pkg_check_modules(CURL REQUIRED libcurl)
pkg_check_modules(CJSON REQUIRED libcjson)
add_executable(rupeesense main.c)
target_include_directories(rupeesense PRIVATE ${MHD_INCLUDE_DIRS} ${CURL_INCLUDE_DIRS} ${CJSON_INCLUDE_DIRS})
target_link_directories(rupeesense PRIVATE ${MHD_LIBRARY_DIRS} ${CURL_LIBRARY_DIRS} ${CJSON_LIBRARY_DIRS})
target_link_libraries(rupeesense PRIVATE ${MHD_LIBRARIES} ${CURL_LIBRARIES} ${CJSON_LIBRARIES})
